if not math.clamp then
    function math.clamp(v, mn, mx) return math.max(mn, math.min(mx, v)) end
end
if not math.round then
    function math.round(n) return math.floor(n + 0.5) end
end
if not table.find then
    function table.find(t, v)
        for i, x in ipairs(t) do if x == v then return i end end
        return nil
    end
end
if not string.split then
    function string.split(s, sep)
        local out = {}
        for token in string.gmatch(s, "([^" .. sep .. "]+)") do out[#out + 1] = token end
        return out
    end
end

local function makeDraggable(frame, handle)
    handle = handle or frame
    local dragging = false
    local dragInput, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if isPress(input) then
            dragging = true
            dragInput = input
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input == dragInput or isMove(input)) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function newPage(name)
    local p = new("Frame", {
        Parent = pageHolder,
        Size = UDim2.new(1, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        BackgroundTransparency = 1,
        Visible = false,

    new("UIListLayout", { Parent = p, Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder })
    new("UIPadding", { Parent = p, PaddingTop = UDim.new(0, 2), PaddingBottom = UDim.new(0, 12), PaddingRight = UDim.new(0, 8) })
    pages[name] = p
    return p
end

local function updateScrollLimit()
    local current = nil
    for _, p in pairs(pages) do
        if p.Visible then current = p; break end
    end
    if not current then pageHolder.CanvasSize = UDim2.new(0, 0, 0, 0); return end
    local layout = current:FindFirstChildOfClass("UIListLayout")
    local padding = current:FindFirstChildOfClass("UIPadding")
    if not layout then pageHolder.CanvasSize = UDim2.new(0, 0, 0, 0); return end
    local h = layout.AbsoluteContentSize.Y
    if padding then h = h + padding.PaddingTop.Offset + padding.PaddingBottom.Offset end
    pageHolder.CanvasSize = UDim2.new(0, 0, 0, h)
end

local function showPage(name)
    for n, p in pairs(pages) do
        if n == name then p.Visible = true else p.Visible = false end
    end
    for n, b in pairs(navButtons) do
        if n == name then
            b.BackgroundColor3 = T.accent
            b.TextColor3 = T.bg
        else
            b.BackgroundColor3 = T.sidebar
            b.TextColor3 = T.text
        end
    end
    navTitle.Text = "Navegando por " .. name
    pageHolder.CanvasPosition = Vector2.new(0, 0)
    task.wait()
    updateScrollLimit()
end

local function attachSliderDrag(track, fill, valueLbl, min, max, step, isFloat, cb, formatFn)
    local dragging = false
    local function apply(px)
        local rel = math.clamp((px - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        local raw = min + rel * (max - min)
        local v = math.floor(raw / step + 0.5) * step
        v = math.clamp(v, min, max)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        if formatFn then valueLbl.Text = formatFn(v)
        elseif isFloat then valueLbl.Text = string.format("%.2f", v)
        else valueLbl.Text = tostring(v) end
        cb(v)
    end
    track.InputBegan:Connect(function(input)
        if isPress(input) then
            dragging = true
            pageHolder.ScrollingEnabled = false
            apply(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and isMove(input) then apply(input.Position.X) end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if isRelease(input) then
            dragging = false
            pageHolder.ScrollingEnabled = true
        end
    end)
end

local function hasLOS(char)
    local myChar = LP.Character
    if not myChar or not char then return false end
    local myHead = myChar:FindFirstChild("Head")
    local tHead = char:FindFirstChild("Head")
    if not myHead or not tHead then return false end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { myChar, char }
    params.IgnoreWater = true
    local result = Workspace:Raycast(myHead.Position, tHead.Position - myHead.Position, params)
    return result == nil
end

local function screenDistFromCenter(pos)
    local vp = Camera.ViewportSize
    local dx = pos.X - vp.X / 2
    local dy = pos.Y - vp.Y / 2
    return math.sqrt(dx*dx + dy*dy)
end

local function entryStillValid(e)
    if not e then return false end
    local plr = e.plr
    if not plr or not plr.Parent then return false end
    local char = plr.Character
    if char ~= e.char or not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    local part = char:FindFirstChild(CONFIG.AIM_PART) or char:FindFirstChild("Head")
    if not part then return false end
    local pos, on = Camera:WorldToViewportPoint(part.Position)
    if not on then return false end
    if screenDistFromCenter(pos) > CONFIG.FOV * 1.5 then return false end
    e.part = part
    e.hum = hum
    return true
end

local function collectTargets()
    local list = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and isEnemy(plr) then
            local char = plr.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if char and hum and hum.Health > 0 then
                local part = char:FindFirstChild(CONFIG.AIM_PART) or char:FindFirstChild("Head")
                if part then
                    local sp, on = Camera:WorldToViewportPoint(part.Position)
                    if on then
                        local sd = screenDistFromCenter(sp)
                        if sd <= CONFIG.FOV then
                            local wp = Camera.CFrame.Position
                            local wd = (part.Position - wp).Magnitude
                            if wd <= CONFIG.AIM_MAX_DIST then
                                local visible = true
                                if CONFIG.AIM_WALL_CHECK then visible = hasLOS(char) end
                                if visible then
                                    list[#list + 1] = {
                                        plr = plr, char = char, hum = hum, part = part,
                                        screenDist = sd, worldDist = wd,
                                    }
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return list
end

local function pickTarget()
    if CONFIG.AIM_LOCK and aimLocked and entryStillValid(aimLocked) then
        return aimLocked
    end
    local list = collectTargets()
    if #list == 0 then
        aimLocked = nil
        return nil
    end
    if CONFIG.AIM_SELECT == "Distância 3D" then
        table.sort(list, function(a, b) return a.worldDist < b.worldDist end)
    elseif CONFIG.AIM_SELECT == "Vida baixa" then
        table.sort(list, function(a, b)
            return (a.hum.Health / math.max(a.hum.MaxHealth, 1)) < (b.hum.Health / math.max(b.hum.MaxHealth, 1))
        end)
    else
        table.sort(list, function(a, b) return a.screenDist < b.screenDist end)
    end
    local best = list[1]
    if CONFIG.AIM_LOCK then aimLocked = best end
    return best
end

local function predictPos(e)
    local pos = e.part.Position
    if not CONFIG.AIM_PREDICT then return pos end
    local hrp = e.char:FindFirstChild("HumanoidRootPart")
    local vel = hrp and hrp.AssemblyLinearVelocity or Vector3.zero
    local dist = (pos - Camera.CFrame.Position).Magnitude
    local speed = math.max(CONFIG.AIM_BULLET_SPEED, 50)
    local t = dist / speed
    local horiz = Vector3.new(vel.X, 0, vel.Z) * t
    local vert = vel.Y * t + 0.5 * Workspace.Gravity * t * t
    local extra = (horiz + Vector3.new(0, vert, 0)) * CONFIG.AIM_PREDICT_STRENGTH
    return pos + extra
end

local function applyAim(e)
    local aimPos = predictPos(e)
    local camPos = Camera.CFrame.Position
    local desired = CFrame.new(camPos, aimPos)
    local s = CONFIG.AIM_SMOOTH
    if s >= 1 then
        Camera.CFrame = desired
    else
        Camera.CFrame = Camera.CFrame:Lerp(desired, math.clamp(s, 0.01, 1))
    end
end

local function hazeAimTick(dt)
    if not gui.Parent then return end
    if CONFIG.AIMBOT and aimActive() then
        local target = pickTarget()
        if target then applyAim(target) end
    else
        aimLocked = nil
    end
    if CONFIG.PRECISION then
        local savedFov, savedPart = CONFIG.FOV, CONFIG.AIM_PART
        CONFIG.FOV = CONFIG.PRECISION_RANGE
        CONFIG.AIM_PART = "Head"
        local t = pickTarget()
        if t then applyAim(t) end
        CONFIG.FOV, CONFIG.AIM_PART = savedFov, savedPart
    end
end

local bindOk = pcall(function()
    RunService:BindToRenderStep("HazeAim", Enum.RenderPriority.Camera.Value + 1, hazeAimTick)
end)

if not bindOk then
    task.spawn(function()
        while gui.Parent do
            pcall(hazeAimTick, 0)
            RunService.RenderStepped:Wait()
        end
    end)
end

    end

    pcall(function()

        vim:SendMouseButtonEvent(0, 0, 0, true, game, 1)
        vim:SendMouseButtonEvent(0, 0, 0, false, game, 1)
    end)

local function autoFireTick()
    if not CONFIG.AUTO_FIRE then return end
    if tick() - lastShot < CONFIG.AUTO_FIRE_DELAY then return end

    local temAlvo = false
    if CONFIG.AIMBOT and aimActive() then
        if CONFIG.AUTO_FIRE_SÓ_LOCK then
            temAlvo = aimLocked ~= nil
        else
            temAlvo = true
        end
    end
    if CONFIG.AUTO_FIRE_PRECISION and CONFIG.PRECISION then
        temAlvo = true
    end

    if temAlvo then
        lastShot = tick()
        disparar()
    end
end

RunService.RenderStepped:Connect(function()
    pcall(autoFireTick)
end)

local savedWeaponData = {}
local function getEquippedTool()
    local char = LP.Character
    if not char then return nil end
    for _, c in ipairs(char:GetChildren()) do
        if c:IsA("Tool") then return c end
    end
    return nil
end

local function applyBigWeapon(tool, mult)
    if not tool then return end
    savedWeaponData[tool] = savedWeaponData[tool] or {}
    for _, obj in ipairs(tool:GetDescendants()) do
        if obj:IsA("BasePart") then
            if not savedWeaponData[tool][obj] then savedWeaponData[tool][obj] = obj.Size end
            obj.Size = savedWeaponData[tool][obj] * mult
        elseif obj:IsA("SpecialMesh") or obj:IsA("BlockMesh") or obj:IsA("CylinderMesh") then
            if not savedWeaponData[tool][obj] then savedWeaponData[tool][obj] = obj.Scale end
            obj.Scale = savedWeaponData[tool][obj] * mult
        elseif obj:IsA("MeshPart") then
            if not savedWeaponData[tool][obj] then savedWeaponData[tool][obj] = obj.Size end
            obj.Size = savedWeaponData[tool][obj] * mult
        end
    end
end

local function restoreBigWeapon(tool)
    if not tool then return end
    local s = savedWeaponData[tool]
    if not s then return end
    for obj, ov in pairs(s) do
        if obj and obj.Parent and typeof(ov) == "Vector3" then
            if obj:IsA("BasePart") or obj:IsA("MeshPart") then obj.Size = ov
            else obj.Scale = ov end
        end
    end
    savedWeaponData[tool] = nil
end

local function restoreAllWeapons()
    for t, _ in pairs(savedWeaponData) do restoreBigWeapon(t) end
end

task.spawn(function()
    local lastTool = nil
    while gui.Parent do
        if CONFIG.BIG_WEAPON then
            local tool = getEquippedTool()
            if tool ~= lastTool then
                if lastTool then restoreBigWeapon(lastTool) end
                lastTool = tool
            end
            if tool then applyBigWeapon(tool, CONFIG.BIG_WEAPON_SIZE) end
        else
            if lastTool then restoreBigWeapon(lastTool); lastTool = nil end
        end
        task.wait(0.2)
    end
end)

local function updateArrow(a, px, py, ang, sz, cor)
    local tx = px + math.cos(ang) * sz
    local ty = py + math.sin(ang) * sz
    local perpX = -math.sin(ang); local perpY = math.cos(ang)
    local bx = px - math.cos(ang) * sz * 0.3
    local by = py - math.sin(ang) * sz * 0.3
    local b1x = bx + perpX * sz * 0.6; local b1y = by + perpY * sz * 0.6
    local b2x = bx - perpX * sz * 0.6; local b2y = by - perpY * sz * 0.6
    local function line(f, x1, y1, x2, y2)
        local dx = x2 - x1; local dy = y2 - y1
        local len = math.sqrt(dx*dx + dy*dy)
        local cx = (x1+x2)/2; local cy = (y1+y2)/2
        local rot = math.deg(math.atan2(dy, dx))
        f.Position = UDim2.new(0, cx, 0, cy)
        f.Size = UDim2.new(0, len, 0, 2)
        f.Rotation = rot; f.BackgroundColor3 = cor
    end
    line(a.l1, tx, ty, b1x, b1y)
    line(a.l2, b1x, b1y, b2x, b2y)
    line(a.l3, b2x, b2y, tx, ty)
    a.parent.Visible = true
end

        local showBox = CONFIG.ESP_BOX and shouldShow(plr, CONFIG.ESP_BOX_TEAM)

        local showSkel = CONFIG.ESP_SKELETON and shouldShow(plr, CONFIG.ESP_SKELETON_TEAM)
        local showMark = CONFIG.ESP_MARK and shouldShow(plr, CONFIG.ESP_MARK_TEAM)
        local showHealth = CONFIG.ESP_HEALTH and shouldShow(plr, CONFIG.ESP_HEALTH_TEAM)
        local showNome = CONFIG.ESP_NOMES and shouldShow(plr, CONFIG.ESP_NOMES_TEAM)

        local char = plr.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not char or not hum or hum.Health <= 0 then
            showBox, show360, showSkel, showMark, showHealth, showNome = false, false, false, false, false, false
        end
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local head = char and char:FindFirstChild("Head")

        if showBox and hrp and head then
            local hp, on1 = Camera:WorldToViewportPoint(head.Position + Vector3.new(0,0.5,0))
            local fp, on2 = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0,3,0))
            if on1 and on2 then
                local h = math.abs(fp.Y - hp.Y)
                local w = h * 0.55
                local x1 = hp.X - w/2; local x2 = hp.X + w/2
                local y1 = math.min(hp.Y, fp.Y); local y2 = math.max(hp.Y, fp.Y)
                local t = CONFIG.ESP_BOX_THICK
                local col = CONFIG.ESP_BOX_COLOR
                d.box.top.Size = UDim2.new(0, w, 0, t); d.box.top.Position = UDim2.new(0, x1, 0, y1); d.box.top.BackgroundColor3 = col; d.box.top.Visible = true
                d.box.bottom.Size = UDim2.new(0, w, 0, t); d.box.bottom.Position = UDim2.new(0, x1, 0, y2); d.box.bottom.BackgroundColor3 = col; d.box.bottom.Visible = true
                d.box.left.Size = UDim2.new(0, t, 0, h); d.box.left.Position = UDim2.new(0, x1, 0, y1); d.box.left.BackgroundColor3 = col; d.box.left.Visible = true
                d.box.right.Size = UDim2.new(0, t, 0, h); d.box.right.Position = UDim2.new(0, x2, 0, y1); d.box.right.BackgroundColor3 = col; d.box.right.Visible = true
            else
                for _, s in ipairs({"top","bottom","left","right"}) do d.box[s].Visible = false end
            end
        else
            for _, s in ipairs({"top","bottom","left","right"}) do if d.box[s] then d.box[s].Visible = false end end
        end

        if showNome and head then
            local hp, on1 = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 3, 0))
            if on1 then
                d.nome.Visible = true
                d.nome.Position = UDim2.new(0, hp.X - 60, 0, hp.Y)
                d.nome.Text = plr.Name
            else
                d.nome.Visible = false
            end
        elseif d.nome then
            d.nome.Visible = false
        end

        if d.health.bg then
            if showHealth and hrp and head then
                local hp, on1 = Camera:WorldToViewportPoint(Vector3.new(hrp.Position.X, head.Position.Y + 1, hrp.Position.Z))
                local fp, on2 = Camera:WorldToViewportPoint(Vector3.new(hrp.Position.X, hrp.Position.Y - 3, hrp.Position.Z))
                if on1 and on2 then
                    local topY = hp.Y
                    local botY = fp.Y
                    local height = math.max(math.abs(botY - topY), 20)
                    local centerY = (topY + botY) / 2
                    local bg = d.health.bg; local fill = d.health.fill
                    local pw, ph = 4, height
                    local posX, posY = hp.X, centerY
                    if CONFIG.ESP_HEALTH_POS == "Esquerda" then posX = hp.X - 30 - 6
                    elseif CONFIG.ESP_HEALTH_POS == "Direita" then posX = hp.X + 30 + 6
                    elseif CONFIG.ESP_HEALTH_POS == "Acima" then posX = hp.X; posY = topY - 12; pw = 50; ph = 4
                    elseif CONFIG.ESP_HEALTH_POS == "Abaixo" then posX = hp.X; posY = botY + 12; pw = 50; ph = 4 end
                    bg.Size = UDim2.new(0, pw, 0, ph)
                    bg.Position = UDim2.new(0, posX, 0, posY)
                    bg.AnchorPoint = Vector2.new(0.5, 0.5)
                    if CONFIG.ESP_HEALTH_POS == "Esquerda" or CONFIG.ESP_HEALTH_POS == "Direita" then
                        fill.Size = UDim2.new(1, 0, hum.Health / math.max(hum.MaxHealth, 1), 0)
                        fill.Position = UDim2.new(0, 0, 1, 0); fill.AnchorPoint = Vector2.new(0, 1)
                    else
                        fill.Size = UDim2.new(hum.Health / math.max(hum.MaxHealth, 1), 0, 1, 0)
                        fill.Position = UDim2.new(0, 0, 0, 0); fill.AnchorPoint = Vector2.new(0, 0)
                    end
                    fill.BackgroundColor3 = healthColor(hum.Health / math.max(hum.MaxHealth, 1))
                    bg.Visible = true
                else
                    d.health.bg.Visible = false
                end
            else
                d.health.bg.Visible = false
            end
        end

        if d.arrow then
            if show360 and hrp then
                local vp = Camera.ViewportSize
                local cx = vp.X / 2; local cy = vp.Y / 2
                local tp, on = Camera:WorldToViewportPoint(hrp.Position)
                if not on then
                    local dx = tp.X - cx; local dy = tp.Y - cy
                    local ang = math.atan2(dy, dx)
                    local r = CONFIG.ESP_360_RADIUS
                    local ax = cx + math.cos(ang) * r
                    local ay = cy + math.sin(ang) * r
                    updateArrow(d.arrow, ax, ay, ang, CONFIG.ESP_360_SIZE, CONFIG.ESP_360_COLOR)
                else
                    d.arrow.parent.Visible = false
                end
            else
                d.arrow.parent.Visible = false
            end
        end

        if d.mark then
            if showMark and head then
                local hp, on = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 1.5, 0))
                if on then
                    d.mark.Size = UDim2.new(0, 12, 0, 12)
                    d.mark.Position = UDim2.new(0, hp.X - 6, 0, hp.Y - 6)
                    d.mark.BackgroundColor3 = CONFIG.ESP_MARK_COLOR
                    d.mark.Visible = true
                else
                    d.mark.Visible = false
                end
            else
                d.mark.Visible = false
            end
        end

        if hum then
            local bones = getBoneList(hum)
            for i, line in pairs(d.lines) do
                local b = bones[i]
                if b and showSkel then
                    local a = char:FindFirstChild(b[1])
                    local c = char:FindFirstChild(b[2])
                    if a and c then
                        local pA, oA = Camera:WorldToViewportPoint(a.Position)
                        local pB, oB = Camera:WorldToViewportPoint(c.Position)
                        if oA and oB then
                            local vA = Vector2.new(pA.X, pA.Y)
                            local vB = Vector2.new(pB.X, pB.Y)
                            local diff = vB - vA
                            local len = diff.Magnitude
                            local center = (vA + vB) / 2
                            local ang = math.deg(math.atan2(diff.Y, diff.X))
                            line.Position = UDim2.new(0, center.X, 0, center.Y)
                            line.Size = UDim2.new(0, len, 0, 1)
                            line.Rotation = ang
                            line.BackgroundColor3 = CONFIG.ESP_SKELETON_COLOR
                            line.Visible = true
                        else
                            line.Visible = false
                        end
                    else
                        line.Visible = false
                    end
                else
                    line.Visible = false
                end
            end
        else
            for _, l in pairs(d.lines) do l.Visible = false end
        end
    end
end

local function updateESPTargets()
    local needESP = CONFIG.ESP_BOX or CONFIG.ESP_360 or CONFIG.ESP_SKELETON or CONFIG.ESP_MARK or CONFIG.ESP_HEALTH or CONFIG.ESP_NOMES
    for plr, _ in pairs(espData) do
        if not plr.Parent or (not needESP) then destroyESPElements(plr) end
    end
    if needESP then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character and not espData[plr] then createESPElements(plr) end
        end
    end
end

        RunService.RenderStepped:Wait()
    end
end)

local function refreshRows()
    for plr, r in pairs(rowCache) do
        if not plr.Parent then r.row:Destroy(); rowCache[plr] = nil end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if not rowCache[plr] then rowCache[plr] = makeRow(plr) end
        local r = rowCache[plr]
        local ok, ping = pcall(function() return plr:GetNetworkPing() * 1000 end)
        r.ping.Text = (ok and ping and ping > 0) and (math.floor(ping) .. "ms") or "—"
        local hum = getHumanoid(plr)
        if hum then
            r.vida.Text = math.floor(hum.Health) .. "/" .. math.floor(hum.MaxHealth)
            r.vida.TextColor3 = healthColor(hum.Health / math.max(hum.MaxHealth, 1))
        else
            r.vida.Text = "—"; r.vida.TextColor3 = T.dim
        end
        if plr.Team then
            r.time.Text = plr.Team.Name
            r.time.TextColor3 = plr.Team.TeamColor and plr.Team.TeamColor.Color or T.dim
        else
            r.time.Text = "—"; r.time.TextColor3 = T.dim
        end
    end
end
refreshRows()

local function refreshStats()
    local k = getLeaderstat(LP, KILL_ALIASES)
    local d = getLeaderstat(LP, DEATH_ALIASES)
    local w = getLeaderstat(LP, WIN_ALIASES)
